import os from telegram import Update from telegram.ext import Application, CommandHandler, ContextTypes async def start(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text( "👋 Welcome to PR02C0L BOT!\n\n" "Your bot is online and working. ✅" ) async def help_command(update: Update, context: ContextTypes.DEFAULT_TYPE): await update.message.reply_text( "🤖 PR02C0L BOT Help\n\n" "/start - Start the bot\n" "/help - Show this help message" ) def main(): token = os.getenv("BOT_TOKEN") if not token: print("BOT_TOKEN is not set!") return app = Application.builder().token(token).build() app.add_handler(CommandHandler("start", start)) app.add_handler(CommandHandler("help", help_command)) print("PR02C0L BOT is starting...") app.run_polling() if __name__ == "__main__": main()
